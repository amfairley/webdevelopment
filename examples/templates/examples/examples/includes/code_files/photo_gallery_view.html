from django.shortcuts import render
from django.conf import settings


def gallery(request):
    '''Return the gallery page'''
    # Create a list of image dictionaries
    # Each image dictionary has a local route, deployed route and alt text
    image_filenames = [
        {
            'local': 'media/images/image1.webp',
            'host': 'onlinehosteswebsite/image1.webp',
            'alt': 'Image 1 alt text'
        },
        {
            'local': 'media/images/image2.webp',
            'host': 'onlinehosteswebsite/image2.webp',
            'alt': 'Image 2 alt text',
        },
    ]
    context = {
        'MEDIA_URL': settings.MEDIA_URL,
        'DEBUG': settings.DEBUG,
        # Images
        'images': image_filenames,
    }
    return render(
        request,
        'gallery/gallery.html',
        context,
    )
